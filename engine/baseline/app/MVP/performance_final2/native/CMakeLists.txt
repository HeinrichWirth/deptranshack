cmake_minimum_required(VERSION 3.18)
project(c4_sprint LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_c4_sprint kernel.cpp)
target_compile_features(_c4_sprint PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(_c4_sprint PRIVATE /O2 /fp:strict /Qvec-report:2)
else()
 target_compile_options(_c4_sprint PRIVATE -O3 -fno-fast-math -ffp-contract=off -fopt-info-vec-optimized)
 if(HOST_OPTIMIZED)
  target_compile_options(_c4_sprint PRIVATE -march=native)
 endif()
endif()
install(TARGETS _c4_sprint LIBRARY DESTINATION MVP/performance_final2/native RUNTIME DESTINATION MVP/performance_final2/native)
