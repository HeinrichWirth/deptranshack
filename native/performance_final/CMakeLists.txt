cmake_minimum_required(VERSION 3.18)
project(final_c4_native LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_final_c4_native kernel.cpp)
target_compile_features(_final_c4_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(_final_c4_native PRIVATE /O2 /fp:strict)
else()
  target_compile_options(_final_c4_native PRIVATE -O3 -fno-fast-math -ffp-contract=off)
endif()
install(TARGETS _final_c4_native LIBRARY DESTINATION MVP/performance_final/native RUNTIME DESTINATION MVP/performance_final/native)
